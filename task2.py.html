from module_ass5 import *
from scipy.integrate import solve_ivp
t_span = np.array([0, 2])
y0 = np.array([4.0])
sol = solve_ivp(derivative_ode1, t_span, y0)
y_approx = np.delete(sol.y, 0)  # y approximation with the initial value removed
time = np.delete(sol.t, 0)
print(y_approx)
y_exact = np.zeros(len(time))

for i in range(len(sol.t) - 1):
    y_exact[i] = exact_solution_ode1(time[i])
print(y_exact)

mae = mean_absolute_error(y_approx, y_exact)
print(mae)

plt.plot(time, y_exact, label='Exact')
plt.plot(time, y_approx, label='Approximate')
plt.legend()
plt.xlabel('t')
plt.ylabel('y')
plt.title("y versus t using exact and approximate solutions")
plt.show()

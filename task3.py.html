# For 3.1 and 3.2:
y0 = 4.
tspan = [0., 2.]
h = [0.01, 0.025, 0.05, 0.1, 0.2, 0.25]
mae_euler = np.zeros(len(h))
mae_ieuler = np.zeros(len(h))
mae_rk4 = np.zeros(len(h))

for i in range(len(h)):
    timestamp_euler, y_euler = explicit_rk_solver(derivative_ode1, tspan, y0, h[i], alpha_euler, beta_euler, gamma_euler)
    y_approx = y_euler[1:]
    y_exact = exact_solution_ode1(timestamp_euler[1:])
    mae_euler[i] = mean_absolute_error(y_exact, y_approx)

for i in range(len(h)):
    timestamp_ieuler, y_ieuler = explicit_rk_solver(derivative_ode1, tspan, y0, h[i], alpha_ieuler, beta_ieuler, gamma_ieuler)
    y_approx = y_ieuler[1:]
    y_exact = exact_solution_ode1(timestamp_ieuler[1:])
    mae_ieuler[i] = mean_absolute_error(y_exact, y_approx)

for i in range(len(h)):
    timestamp_rk4, y_rk4 = explicit_rk_solver(derivative_ode1, tspan, y0, h[i], alpha_rk4, beta_rk4, gamma_rk4)
    y_approx = y_rk4[1:]
    y_exact = exact_solution_ode1(timestamp_rk4[1:])
    mae_rk4[i] = mean_absolute_error(y_exact, y_approx)

plt.plot(h, mae_euler, label='Euler method')

plt.title("MAE vs step size")
plt.ylabel("Mean Absolute Value")
plt.xlabel("Step Size")

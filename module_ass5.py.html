def exact_solution_ode1(t):
    """ TODO 
    Calculates the exact solution to the initial value problem in (7) 
    ---------- 
 
    Parameter(s) 
    ---------- 
    t: integer/float/array-like 
        Time at which the solution of y is to be calculated 
 
    Returns 
    ------- 
    y_exact: integer/float/array-like 
        Exact solution of the initial value problem for the input value t 
    """

    # Converting the input into a NumPy array.
    t = np.asarray(t)

    # Calculating the exact solution to the input with the particular solution equation.
    y_exact = np.exp(-5 * t) * (2 * t + 4)

    # Convert back to a float if the solution is a NumPy float.
    if type(y_exact) is np.float64:
        y_exact = float(y_exact)
    return y_exact


def mean_absolute_error(y_exact, y_approx):
    """ TODO 
    Calculates the mean absolute error for an approximate solution, given the exact solution. 
    ---------- 
 
    Parameter(s) 
    ---------- 
    y_exact: 1D NumPy array 
        Exact solution (excluding initial condition) 
    y_approx: 1D NumPy array 
        Numerical approximation to the exact solution above (excluding initial condition) 
 
    Returns 
    ------- 
    mae: float 
        Mean absolute error in the numerical solution 
 
    Pre/Postcondition 
    ----------------- 
    -y_exact and y_approx must both be NumPy arrays of the same length 
    """

    # Initialise variable mae
    mae = 0
    n = len(y_exact)

    # Loop through each element in the exact and approximate y solution.
    # Add to mae the absolute difference between approximate and exact solution divided by n for each iteration.
    for i in range(n):
        mae += abs(y_exact[i] - y_approx[i]) / n

    # Convert mae into a float.
    mae = float(mae)
    return mae

def euler_step(f, t, y, h):
    """ 
    Calculate one step of the Euler method. 
 
    Parameters 
    ---------- 
    f : function 
        Derivative function (callable). 
    t : float 
        Independent variable at start of step. 
    y : float 
        Dependent variable at start of step. 
    h : float 
        Step size along independent variable. 
 
    Returns 
    ------- 
    y_new : float 
        Dependent variable at end of step. 
    """
    f0 = f(t, y)
    y_new = y + h * f0
    return y_new


def improved_euler_step(f, t, y, h):
    """ 
    Calculate one step of the Improved Euler method. 
 
    Parameters 
    ---------- 
    f : function 
        Derivative function (callable). 
    t : float 
        Independent variable at start of step. 
    y : float 
        Dependent variable at start of step. 
    h : float 
        Step size along independent variable. 
 
    Returns 
    ------- 
    y_new : float 
        Dependent variable at end of step. 
    """
    f0 = f(t, y)
    f1 = f(t + h, y + h * f0)
    y_new = y + h * 0.5 * (f0 + f1)
    return y_new


def classic_rk4_step(f, t, y, h):
    """ 
    Calculate one step of the Classic RK4 method. 
 
    Parameters 
    ---------- 
    f : function 
        Derivative function (callable). 
    t : float 
        Independent variable at start of step. 
    y : float 
        Dependent variable at start of step. 
    h : float 
        Step size along independent variable. 
 
    Returns 
    ------- 
    y_new : float 
        Dependent variable at end of step. 
    """
    f0 = f(t, y)
    f1 = f(t + h * 0.5, y + h * 0.5 * f0)
    f2 = f(t + h * 0.5, y + h * 0.5 * f1)
    f3 = f(t + h, y + h * f2)
    y_new = y + h * (f0 + 2. * f1 + 2. * f2 + f3) / 6.
    return y_new


def explicit_rk_step(f, t, y, h, alpha, beta, gamma):
    """ TODO 
    Calculate one step of any explicit RK method 
 
    Parameters 
    ---------- 
    f : function 
        Derivative function (callable). 
    t : float 
        Independent variable at start of step. 
    y : float 
        Dependent variable at start of step. 
    h : float 
        Step size along independent variable. 
    alpha: 1D NumPy array 
        Weights from the Butcher Tableau of the method 
    beta: 1D NumPy array 
        Nodes from the Butcher Tableau of the method 
    gamma: 1D NumPy array 
        RK matrix from the Butcher Tableau of the method 
 
    Returns 
    ------- 
    y_new : float 
        Dependent variable at end of step. 
    """
    # Initialising our new y-value as the y given in the input.
    y_new = y

    # seeing what order of method was chosen by user by looking at the size of alpha array
    order = len(alpha)

    # Initialising an array to store derivative evaluations
    # the number of derivative evaluations should be include to the order number.
    estimation = [0] * order

    # Calculate each derivative evaluation with corresponding beta and gamma values
    for i in range(order):
        estimation[i] = f((t + beta[i] * h), (y + h * gamma[i][i-1] * estimation[i-1]))

    # Multiplying our derivation evaluations by the step size and the corresponding alpha value.
    # Then adding it onto our original y value which was provided in the input
    for i in range(order):
        y_new += h * estimation[i] * alpha[i]

    return y_new


def explicit_rk_solver(f, tspan, y0, h, alpha, beta, gamma):
    """ TODO 
    Calculates the dependent variable for a certain step size over a certain timespan. 
 
    Parameters 
    ---------- 
    f : function 
        Derivative function (callable). 
    tspan : list 
        Two floats representing the start and end values of the independent variable 
    y0 : float 
        Initial condition for the dependent variable. 
    h : float 
        Step size along independent variable. 
    alpha: 1D NumPy array 
        Weights from the Butcher Tableau of the method 
    beta: 1D NumPy array 
        Nodes from the Butcher Tableau of the method 
    gamma: 1D NumPy array 
        RK matrix from the Butcher Tableau of the method 
 
    Returns 
    ------- 
    t: 1D NumPy array 
        Values of the independent variable for each step. 
    y: 1D NumPy array 
        Value of dependent variable for each step. 
 
    Pre/Postconditions 
    ------------------ 
    -Input tspan array must be of length 2, t[1]<t[2] 
    """
    # Initialising initial dependent/independent variables,
    # and arrays to store values of dependent/independent variables for each step ahead.
    t = tspan[0]
    timestamp = [t]
    dep_var = [y0]
    y = y0

    # The number of values for each of our variable is equal to the difference
    # in the dependent variable divided by step size plus one
    array_size = (tspan[1] - tspan[0])/h + 1

    # Calculate each value of our independent/dependent variable until the length of our array reaches the
    # supposed size
    while len(timestamp) < array_size:
        y_intermediate = explicit_rk_step(f, t, y, h, alpha, beta, gamma)
        print(y_intermediate)
        dep_var.append(y_intermediate)
        y = y_intermediate
        t += h
        timestamp.append(t)
    timestamp = np.asarray(timestamp)
    dep_var = np.asarray(dep_var)
    return timestamp, dep_var
